  Set-Mailbox <UPN> -ElcProcessingDisabled $false

  Set-MailboxAutoReplyConfiguration -Identity <UPN> -AutoReplyState Enabled -InternalMessage "<msg>" -ExternalMessage "<msg>"

[GROUPES]

================================================================================
  NOTES — PARAMÈTRES
================================================================================

  <UPN>      = Adresse email / identifiant Entra ID de l'utilisateur
               ex. john.doe@contoso.com

  <boite>    = UPN ou alias de la boîte cible (partagée ou utilisateur)
               ex. contact@contoso.com

  <email>    = Adresse email d'un expéditeur ou destinataire (MessageTrace)
               ex. expediteur@externe.com

  <GroupId>  = GUID du groupe Entra ID
               (récupérable via Get-MgGroup -Filter "displayName eq '<nom>'")

  <ObjectId> = GUID de l'objet Entra ID (utilisateur, appareil…)
               (récupérable via Get-MgUser -UserId <UPN> | Select-Object Id)

  <UserId>   = GUID de l'utilisateur Entra ID (même valeur que ObjectId)

  <SkuId>    = GUID de la licence
               (récupérable via Get-MgSubscribedSku | Select-Object SkuId, SkuPartNumber)

  <id>       = Identifiant retourné par la commande précédente
               ex. Id de quarantaine, Id de méthode MFA, Id d'appareil Intune

  <nom>      = Nom du service Windows ou du processus
               ex. Spooler, W32Time, Teams, OneDrive

  <host>     = Nom d'hôte ou adresse IP de destination
               ex. google.com  /  192.168.1.1  /  smtp.office365.com

  <domaine>  = Nom de domaine DNS
               ex. contoso.com  /  fabrikam.fr

  <port>     = Numéro de port TCP/UDP
               ex. 443 (HTTPS)  /  25 (SMTP)  /  587 (SMTP auth)  /  3389 (RDP)

  <ip>       = Adresse IPv4 ou IPv6
               ex. 192.168.1.100  /  10.0.0.1

  <date>     = Date au format MM/DD/YYYY (requis par Get-MessageTrace)
               ex. 05/20/2026

  <msg>      = Texte du message d'absence (OOF)
               ex. "Je suis absent jusqu'au 30/05. Contactez support@contoso.com."

  <chemin>   = Chemin d'accès fichier ou dossier
               ex. C:\Temp\rapport.html  /  \\serveur\partage\dossier

================================================================================
  CORRECTIONS APPORTÉES
================================================================================

  1. Connect-ExchangeOnline
     → Ajout de -UserPrincipalName <UPN> (auth moderne, sans mot de passe en clair)

  2. Get-MailboxStatistics -Archive -Identity <UPN> TotalItemSize
     → Corrigé en : ... | Select-Object TotalItemSize
       (TotalItemSize n'est pas un paramètre valide de la cmdlet)

  3. wuauclt /detectnow
     → Marqué [DÉPRÉCIÉ] sur Windows 10/11. Utiliser UsoClient StartScan à la place.
       wuauclt fonctionne encore mais ne garantit pas le déclenchement sur Win10+.
